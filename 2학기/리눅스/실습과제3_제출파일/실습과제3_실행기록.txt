9. 자식 디렉터리가 가장 많은 디렉터리

find . -mindepth 1 -type d -print0 | while IFS= read -r -d '' d; do
  count=$(find "$d" -mindepth 1 -maxdepth 1 -type d | wc -l)
  echo "$count $d"
done | sort -nr
